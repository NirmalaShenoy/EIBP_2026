// Define a structure for the global processed labels list
typedef struct ProcessedLabel {
    char *label;
    struct ProcessedLabel *next;
} ProcessedLabel;

ProcessedLabel *processedLabels = NULL;
pthread_mutex_t processedLabelsMutex = PTHREAD_MUTEX_INITIALIZER;

// Function to check if a label has been processed
bool is_label_processed(const char *label) {
    pthread_mutex_lock(&processedLabelsMutex);

    ProcessedLabel *current = processedLabels;
    while (current != NULL) {
        if (strcmp(current->label, label) == 0) {
            pthread_mutex_unlock(&processedLabelsMutex);
            return true;
        }
        current = current->next;
    }

    pthread_mutex_unlock(&processedLabelsMutex);
    return false;
}

// Function to add a label to the processed list
void add_processed_label(const char *label) {
    pthread_mutex_lock(&processedLabelsMutex);

    ProcessedLabel *newLabel = malloc(sizeof(ProcessedLabel));
    newLabel->label = strdup(label);
    newLabel->next = processedLabels;
    processedLabels = newLabel;

    pthread_mutex_unlock(&processedLabelsMutex);
}

// Function to remove a label from the processed labels list
void remove_processed_label(const char *label) {
    pthread_mutex_lock(&processedLabelsMutex);

    ProcessedLabel *current = processedLabels;
    ProcessedLabel *previous = NULL;

    while (current != NULL) {
        if (strcmp(current->label, label) == 0) {
            // Found the label, remove it from the list
            if (previous == NULL) {
                // Removing the head
                processedLabels = current->next;
            } else {
                previous->next = current->next;
            }
            free(current->label);
            free(current);
            pthread_mutex_unlock(&processedLabelsMutex);
            return;
        }
        previous = current;
        current = current->next;
    }

    pthread_mutex_unlock(&processedLabelsMutex);
}


//Ignores control characters such as SOH, STX, BEL ,etc
void filterControlChars(char* str) {
	int len = strlen(str);
	for (int i = 0; i < len; i++) {
		if (str[i] < 32) {
			continue;
		}
		// printf("%c", str[i]);
	}
}

// calculates the time difference in micro seconds
double time_diff(struct timeval before, struct timeval after)
{
	double x_ms, y_ms, diff;
	x_ms = (double)before.tv_sec * 1000000 + (double)before.tv_usec;
	y_ms = (double)after.tv_sec * 1000000 + (double)after.tv_usec;
	diff = (double)y_ms - (double)x_ms;
	return diff;
}

/********************************************************
* setInterfaces() //interfaceList is being set by this function
* method to set active interfaces in a interfaceList
* @return status (int) - method return value
********************************************************/
int setInterfaces() {

    if (isEnvSet() != 0)
        setControlIF();
    struct ifaddrs* ifaddr, * ifa;
    int family;
    int countInterface = 0;
    char temp[20];
	char lostInterface[20] = "";
    if (getifaddrs(&ifaddr) == -1) { 
	/*The getifaddrs() function creates a linked list of structures describing the
	 network interfaces of the local system, and stores the address of the first item of the list in *ifap.*/
        // errorCount++;
        perror("ERROR: getifaddrs");
        exit(EXIT_FAILURE);
    }
    int init = 0;
    for (ifa = ifaddr; ifa != NULL; ifa = ifa->ifa_next) {
        if (ifa->ifa_addr == NULL)
            continue;
        family = ifa->ifa_addr->sa_family;
        if (family == AF_PACKET) {
            strcpy(temp, ifa->ifa_name);// interface name
           // printf("\n START: Interface: %s, Flags: %d\n", ifa->ifa_name, ifa->ifa_flags); // Print interface name and flags

            if (strcmp(temp, "meas") == 0 || strcmp(temp, "docker0") == 0)  // Ignore the "meas" and "docker0" interfaces
                continue; // Skip processing these interfaces
            
            if ((ifa->ifa_flags & IFF_UP) == 0) { // testing whether interface is up or not
                //if interface is down
                if (down_flag == 0) { // doing it the first time only - NS ?? why
                    //gettimeofday(&down_time, NULL);
                    //d_time = ((double)down_time.tv_sec*1000000 + (double)down_time.tv_usec) / 1000000;
                    printf("\nIF_DIRECT_DOWN:%lf\n", getCurrentTime());
                    down_flag = 1;
					
					strcpy(lostInterface, ifa->ifa_name);
					strcpy(failedInterface,lostInterface);
					

                }
			//printf("\n IN THE LOOP: Interface: %s, Flags: %d, interface size: %d\n", ifa->ifa_name, ifa->ifa_flags, interfaceListSize);
            }
            else {
				
                if ((ifa->ifa_flags & IFF_LOOPBACK) == 0) { // if not loop_back 
                    if (ctrlIFName != NULL) {
                        if (strlen(temp) == strlen(ctrlIFName)) {
                            if ((strncmp(temp, ctrlIFName, strlen(ctrlIFName))
                                != 0)) {
                                countInterface = countInterface + 1;
                                interfaceList[init] = malloc(1 + strlen(temp));
                                strcpy(interfaceList[init], temp);
                                init = init + 1;
                            }
                        }
                        else {
                            countInterface = countInterface + 1;
                            interfaceList[init] = malloc(1 + strlen(temp));
                            strcpy(interfaceList[init], temp);
                            init = init + 1;
                        }
                    }
                    else {
                        countInterface = countInterface + 1;
                        interfaceList[init] = malloc(1 + strlen(temp));
                        strcpy(interfaceList[init], temp);
                        init = init + 1;
                    }
                }
				
            }
        }
    }
    interfaceListSize = countInterface;
	//printf("\nInterfacelist size = %d", interfaceListSize);
	if(strlen(lostInterface) > 0){
		printf("\nFailed interface: %s\n",lostInterface);
		notify_lostmylabels_direct_down(lostInterface);
		deleteStable(lostInterface);
	}
	
    freeifaddrs(ifaddr);
    return 0;
}


/************************************************
* freeInterfaces()- method to free interfaceList
* @return status (int) - method return value
************************************************/
int freeInterfaces() {
	int loopCounter1 = 0;
	for (; loopCounter1 < interfaceListSize; loopCounter1++) {
		free(interfaceList[loopCounter1]);
	}
	interfaceListSize = 0;
	return 0;
}

/************************************************
* convertStringToInteger()
* Method which converts the passed string into integer.
* @return status (int) - method returns integer value of the string.
*************************************************/
int convertStringToInteger(char* num)
{
	int result = 0, i = 0;
	int len = strlen(num);
	for (i = 0; i < len; i++) {
		result = result * 10 + (num[i] - '0');
	}
	//printf("%d", result);
	return result;
}

/*******************************************************
* freqCount(char[],char[])
* method to find count of particular character
* @param str (char[]) - string
* @param search (char[]) - character to find occurrence
* @return count (int) - occurrence count of a character
********************************************************/
int freqCount(char str[], char search) {

	int i, count = 0;
	for (i = 0; str[i] != '\0'; ++i) {
		if (str[i] == search)
			count++;
	}
	return count;
}

/********************************************************** 
*This function is used to populate the failed IP's to the variable "failedEndIPs_head". 
*addr_tuple is being populated by the active interfaces.
*******************************************************/
void checkForLinkFailures(struct addr_tuple* myAddr, int numTierAddr) {
	struct addr_tuple* current = failedEndIPs_head;
	struct addr_tuple* previous = NULL;
	while (current != NULL) { // Check if any failed End IPS became active.
		if (isInterfaceActive(current->ip_addr, current->cidr)) {
			struct addr_tuple* a = (struct addr_tuple*)calloc(1, sizeof(struct addr_tuple));
			strcpy(a->tier_addr, current->tier_addr); // Add into the table as they are active.
			a->if_index = -1; // insert info about index from which the packet has been received from.
			a->isNew = true;
			memcpy(&a->ip_addr, &current->ip_addr, sizeof(struct in_addr));
			a->cidr = current->cidr;
			add_entry_LL(a);
			print_entries_LL();
			struct addr_tuple* freeptr;
			if (current == failedEndIPs_head) {
				failedEndIPs_head = current->next;
				previous = NULL;
				freeptr = current;
				current = NULL;
			}
			else {
				previous->next = current->next;
				freeptr = current;
				current = current->next;
			}
			free(freeptr);
			continue;
		}
		previous = current;
		current = current->next;
	}
	int  i = 0;
	for (; i < numTierAddr; i++) { // and check to see if address is already added to failed IP list or not.
		struct addr_tuple* ptr = failedEndIPs_head;
		bool isInFailedList = false;
		while (ptr != NULL) {
			if ((myAddr[i].ip_addr.s_addr == ptr->ip_addr.s_addr) && (strcmp(myAddr[i].tier_addr, ptr->tier_addr) == 0)) {
				isInFailedList = true;
				break;
			}
			ptr = ptr->next;
		}
		//printf("* %u *",myAddr[i].cidr); // if interface is not active, add to failed IP list.
		if ((!isInFailedList) && (!isInterfaceActive(myAddr[i].ip_addr, myAddr[i].cidr))) {
			struct addr_tuple* temp = (struct addr_tuple*)calloc(1, sizeof(struct addr_tuple));
			memcpy(temp, &myAddr[i], sizeof(struct addr_tuple));
			temp->isNew = true;
			temp->next = NULL;
			delete_entry_LL_IP(myAddr[i].ip_addr);
			if (failedEndIPs_head == NULL) {
				failedEndIPs_head = temp;
				//printf("Will be removing this %s\n", inet_ntoa(failedEndIPs_head->ip_addr));
			}
			else {
				temp->next = failedEndIPs_head;
				failedEndIPs_head = temp;
			}
		}
	}
}

 * Edge Nodes loosing labels

	if (failedEndIPs_head == NULL) {
		printf("\n failedEndIPs_head is NULL\n");
	}
	else {
		struct addr_tuple* current = failedEndIPs_head;
		struct addr_tuple* prev = failedEndIPs_head;

			current = current->next;
		}
	}
}


/**********************************************************
This funtion is used to find all interfaces in a node and check whether interfaces are active
**************************************************************/
bool isInterfaceActive(struct in_addr ip, int cidr) {
	struct ifaddrs* ifaddr, * ifa;
	if (getifaddrs(&ifaddr)) { /* The getifaddrs() function creates a linked list of structures describing the network interfaces of the local system, and stores the address of the first item of the list in *ifap. */
		perror("Error: getifaddrs() Failed\n");
		if (enableLogScreen)
			printf("\n Is Interface Active\n");
		exit(0);
	}
	for (ifa = ifaddr; ifa != NULL; ifa = ifa->ifa_next) {
		if (ifa->ifa_addr == NULL) {
			continue;
		}
		int family;
		family = ifa->ifa_addr->sa_family;
		// populate interface names, if interface is UP and if ethernet interface doesn't belong to control interface and Loopback interface.
		if (family == AF_INET && (strncmp(ifa->ifa_name, "lo", 2) != 0) && (ifa->ifa_flags & IFF_UP) != 0) {

			struct sockaddr_in* ipaddr = ((struct sockaddr_in*)ifa->ifa_addr);
			struct sockaddr_in* subnmask = ((struct sockaddr_in*)ifa->ifa_netmask);

			if (ip.s_addr == (ipaddr->sin_addr.s_addr & subnmask->sin_addr.s_addr)) {
				freeifaddrs(ifaddr);
				return true;
			}
		}
	}
	freeifaddrs(ifaddr);
	return false;
}

				}
			}

			if (failedCount > 0) {

				for (int i = 0; i < failedCount; i++) {

				}
			} else {
				printf("No derived labels found for deletion.\n");
			}
		}


		//printf("\nNeighbour Table after removing the addresses of the unreachable node CURRENT_TIME:%lf",getCurrentTime());

		print_entries_LL();

/****************************************************************
 * notify_lostmyparent()
 * Called when a parent port (match_port==3) fails on a tier-2 node.
 * Steps:
 *   1. Collect and remove own labels on the failed parent port from headTL.
 *   2. Remove parent neighbor entries on the failed port from headHL.
 *   3. For each lost own label, find and delete child-derived labels from headHL.
 *   4. Collect all lost labels (own + child-derived) and send
 *      MESSAGE_TYPE_LABELS_LOST to remaining parent interfaces.
 *****************************************************************/
void notify_lostmyparent(char failedPort[20]) {

    // Step 1: Collect own labels from headTL where port == failedPort, then remove them
    int ownLabelCount = 0;
    char *ownLostLabels[100];
    struct nodeTL *tlCurrent = headTL, *tlPrev = NULL;
    while (tlCurrent != NULL) {
        if (strcmp(tlCurrent->port, failedPort) == 0) {
            if (ownLabelCount < 100) {
                ownLostLabels[ownLabelCount] = malloc(strlen(tlCurrent->tier) + 1);
                strcpy(ownLostLabels[ownLabelCount++], tlCurrent->tier);
            }
            printf("\nRemoving own label %s from headTL (port %s)\n", tlCurrent->tier, failedPort);
            if (tlCurrent == headTL) {
                headTL = tlCurrent->next;
                free(tlCurrent);
                tlCurrent = headTL;
            } else {
                tlPrev->next = tlCurrent->next;
                free(tlCurrent);
                tlCurrent = tlPrev->next;
            }
        } else {
            tlPrev = tlCurrent;
            tlCurrent = tlCurrent->next;
        }
    }
    printf("\nnotify_lostmyparent: %d own labels lost on port %s\n",
           ownLabelCount, failedPort);

    // Step 2: Remove neighbor entries on the failed parent port from headHL
    struct nodeHL *hlCurrent = headHL, *hlPrev = NULL;
    while (hlCurrent != NULL) {
        if (strcmp(hlCurrent->port, failedPort) == 0) {
            printf("Removing neighbor %s from headHL (port %s)\n", hlCurrent->tier, failedPort);
            if (hlCurrent == headHL) {
                headHL = hlCurrent->next;
                free(hlCurrent);
                hlCurrent = headHL;
            } else {
                hlPrev->next = hlCurrent->next;
                free(hlCurrent);
                hlCurrent = hlPrev->next;
            }
        } else {
            hlPrev = hlCurrent;
            hlCurrent = hlCurrent->next;
        }
    }

    // Step 3: Find child labels derived from each lost own label, then remove from headHL
    int childLostCount = 0;
    char *childLostLabels[100];
    for (int i = 0; i < ownLabelCount; i++) {
        char myLabelCopy[50];
        strncpy(myLabelCopy, ownLostLabels[i], sizeof(myLabelCopy) - 1);
        myLabelCopy[sizeof(myLabelCopy) - 1] = '\0';

        char *tokens[4];
        int count = 0;
        char *saveptr;
        char *token = strtok_r(myLabelCopy, ".", &saveptr);
        while (token && count < 4) {
            tokens[count++] = token;
            token = strtok_r(NULL, ".", &saveptr);
        }

        // Own label has form "tier.X.Y" (3 tokens) → derived child labels share middle "X.Y"
        if (count == 3) {
            char middleSegment[20];
            snprintf(middleSegment, sizeof(middleSegment), "%s.%s", tokens[1], tokens[2]);
            getDerivedLabelsFromNeighbourTable(middleSegment, childLostLabels, &childLostCount);
        }
    }

    // Delete child-derived labels from headHL
    if (childLostCount > 0) {
        char dummyPort[5] = "";
        delete_neighbour(childLostLabels, childLostCount, dummyPort);
    }

    // Step 3b: Delete own lost labels from the IP-to-label map (tablehead / entries_LL).
    // Pass myTier=3 to suppress the internal publishIPLabelMap_updated inside
    // delete_entry_LL_Addr_updated — we send the full label set below.
    if (ownLabelCount > 0) {
        char **deletedFromLL;
        int deletedFromLLCount = 0;
        char recvPort[5] = "";
        delete_entry_LL_Addr_updated(ownLostLabels, ownLabelCount, recvPort, 3,
                                     &deletedFromLL, &deletedFromLLCount);
        if (deletedFromLLCount > 0) {
            print_entries_LL();
            for (int i = 0; i < deletedFromLLCount; i++) free(deletedFromLL[i]);
        }
        if (deletedFromLL) free(deletedFromLL);
    }

    // Step 4: Build combined lost-labels array, filter already-processed labels,
    // then send MESSAGE_TYPE_LABELS_LOST to remaining parent ports.
    int totalLostCount = ownLabelCount + childLostCount;
    if (totalLostCount == 0) {
        printf("notify_lostmyparent: no labels to report for port %s\n", failedPort);
        printNeighbourTable(true);
        printMyLabels();
        return;
    }

    char *allLostLabels[totalLostCount];
    for (int i = 0; i < ownLabelCount; i++)
        allLostLabels[i] = ownLostLabels[i];
    for (int i = 0; i < childLostCount; i++)
        allLostLabels[ownLabelCount + i] = childLostLabels[i];

    // Filter out labels already marked as processed to avoid duplicate LABELS_LOST sends
    char *newLabels[totalLostCount];
    int newLabelCount = 0;
    for (int i = 0; i < totalLostCount; i++) {
        if (!is_label_processed(allLostLabels[i])) {
            newLabels[newLabelCount++] = allLostLabels[i];
            add_processed_label(allLostLabels[i]);
        }
    }

    if (newLabelCount == 0) {
        printf("notify_lostmyparent: all labels already processed, not sending LABELS_LOST\n");
        printNeighbourTable(true);
        printMyLabels();
        for (int i = 0; i < ownLabelCount; i++) free(ownLostLabels[i]);
        for (int i = 0; i < childLostCount; i++) free(childLostLabels[i]);
        return;
    }

    // Build payload: [numLabels][len][label]...
    char labelLostPayload[200];
    memset(labelLostPayload, '\0', sizeof(labelLostPayload));
    uint8_t numLabels = (uint8_t)newLabelCount;
    memcpy(labelLostPayload, &numLabels, 1);
    int payloadOffset = 1;
    for (int i = 0; i < newLabelCount; i++) {
        uint8_t labelSize = (uint8_t)strlen(newLabels[i]);
        memcpy(labelLostPayload + payloadOffset, &labelSize, 1);
        payloadOffset += 1;
        memcpy(labelLostPayload + payloadOffset, newLabels[i], labelSize);
        payloadOffset += labelSize;
    }

    printf("\nSending MESSAGE_TYPE_LABELS_LOST with %d labels (own=%d child=%d)\n",
           newLabelCount, ownLabelCount, childLostCount);

    setInterfaces();
    for (int i = 1; i < interfaceListSize; i++) {
        if (strcmp(interfaceList[i], failedPort) == 0) continue;
        if (myTierValue == 3 && strcmp(interfaceList[i], ipPortToSkip) == 0) continue;
        if (matchPort(interfaceList[i]) == 3) {
            printf("\nSending MESSAGE_TYPE_LABELS_LOST to parent on %s at CURRENT_TIME:%lf\n",
                   interfaceList[i], getCurrentTime());
            ctrlLabelSend(MESSAGE_TYPE_LABELS_LOST, interfaceList[i], labelLostPayload);
        }
    }
    interfaceListSize = 0;

    printNeighbourTable(true);
    printMyLabels();

    for (int i = 0; i < ownLabelCount; i++) free(ownLostLabels[i]);
    for (int i = 0; i < childLostCount; i++) free(childLostLabels[i]);
}

// Tejas addition - Rewritten deleteIPLabel (Fix 6)
// Fix 6: Include tier 2 label (newaddress) in delete message along with grandchild labels.
//        Pass myTier=3 to modify_LL_updated to suppress auto-send, build combined array,

        snprintf(newaddress, sizeof(newaddress), "2.%s.%s", myTierAddress + 2, failedEth + 3);
        printf("\n newaddress = %s\n", newaddress);

        int matchCount = 0;
        current = tablehead;

            if (checkIfGrandParent(current->tier_addr, newaddress) && !is_label_processed(current->tier_addr)) {
                matchCount++;

/**********************************************************************************
 * Tejas 
 * if the tierValue is < my TierValue , ignore the message
	else   generate the label based on the interface it recived the message from
	Send the label to the node  
***********************************************************************************/
void addOrUpdateInterfaceLabels(char recvOnEtherPort[5], struct labels* labelList) {
    struct childInterfaceToLabelList* current = headInterfaceLabel;

    // Check if the interface already exists
    while (current != NULL) {
        if (strcmp(current->interface, recvOnEtherPort) == 0) {
            // Free the existing label list before updating
            struct labels* temp = current->labelList;
            while (temp != NULL) {
                struct labels* toFree = temp;
                temp = temp->next;
                free(toFree);
            }
            // Update with the new label list
            current->labelList = labelList;
            return;
        }
        current = current->next;
    }

    // If the interface doesn't exist, create a new node
    struct childInterfaceToLabelList* newNode = (struct childInterfaceToLabelList*)malloc(sizeof(struct childInterfaceToLabelList));
    strcpy(newNode->interface, recvOnEtherPort);
    newNode->labelList = labelList;
    newNode->next = headInterfaceLabel;
    headInterfaceLabel = newNode;
}


void deleteLabelsByInterface(char recvOnEtherPort[5]) {
    struct childInterfaceToLabelList* current = headInterfaceLabel;
    struct childInterfaceToLabelList* prev = NULL;


    while (current != NULL) {
        if (strcmp(current->interface, recvOnEtherPort) == 0) {
            // Print the labels before deleting
            struct labels* temp = current->labelList;
          //  printf("Labels to be deleted for interface %s:\n", recvOnEtherPort);
            while (temp != NULL) {
               // printf("  %s\n", temp->label);  // Assuming the label field is 'label'
                temp = temp->next;
            }

            // Free the label list
            temp = current->labelList;  // Re-initialize temp to free the list
            while (temp != NULL) {
                struct labels* toFree = temp;
                temp = temp->next;
                free(toFree);
            }

            // Remove the interfaceLabels node
            if (prev == NULL) {
                headInterfaceLabel = current->next;
            } else {
                prev->next = current->next;
            }
            free(current);
        //    printf("\nDeleted labels for interface: %s\n", recvOnEtherPort);
            return;
        }
        prev = current;
        current = current->next;
    }

    printf("\nNo labels found for interface: %s\n", recvOnEtherPort);
}

        printf("Label %d: %s\n", i + 1, labels[i]);

        if (!is_label_processed(labels[i])) {
            newLabels[newLabelCount++] = labels[i];
            add_processed_label(labels[i]);
        }
    }

    if (newLabelCount == 0) {
        printf("\nAll labels already processed, ignoring.\n");
        return;
    }

    // Step 1: Delete received labels from IP-to-label map (tablehead / entries_LL).
    // Pass myTier=3 to suppress the internal publishIPLabelMap_updated call inside
    // delete_entry_LL_Addr_updated — we forward ALL received labels below, not just
    // the subset that happened to exist in the IP-to-label map.
    modify_LL_updated(labels, totalEntries, recvOnEtherPort, 3);

    // Step 2: Delete received labels from neighbor table (headHL)
    delete_neighbour(newLabels, newLabelCount, recvOnEtherPort);

    printNeighbourTable(true);
    print_entries_LL();

    // Step 3: Forward MESSAGE_TYPE_PUBLISH_IP_DELETE with ALL labels from LABELS_LOST,
    // regardless of which tables they were found in.
    publishIPLabelMap_updated(newLabels, newLabelCount, 1, recvOnEtherPort, false);
}

void extractLabelSegments(char *label, int *tier, char **middleSegment, char **port) {
    static char labelCopy[50];  // Static buffer to persist across function calls
    strcpy(labelCopy, label);

    char *tokens[4];  // Max of 4 segments
    int count = 0;

    char *saveptr;
    char *token = strtok_r(labelCopy, ".", &saveptr);
    while (token) {
        tokens[count++] = token;
        token = strtok_r(NULL, ".", &saveptr);
    }

    if (count < 3) {  // Must have at least "tier.middle.port"
        *tier = -1;
        *middleSegment = NULL;
        *port = NULL;
        return;
    }

    // Extract tier (first segment)
    *tier = atoi(tokens[0]);

    // Extract port (last segment)
    *port = tokens[count - 1];

    // Extract middle segment correctly
    static char middleBuffer[50];  
    middleBuffer[0] = '\0';  // Clear buffer

    if (count == 3) {  // Case: "tier.middle.port"
        strcpy(middleBuffer, tokens[1]);  // Just the middle part
    } else if (count == 4) {  // Case: "tier.middle1.middle2.port"
        snprintf(middleBuffer, sizeof(middleBuffer), "%s.%s", tokens[1], tokens[2]);
    }

    *middleSegment = middleBuffer;

    // Debug: Print extracted segments
    printf("Original Label: %s | Tier: %d | Middle Segment: %s | Port: %s\n",
           label, *tier,
           *middleSegment ? *middleSegment : "NULL",
           *port ? *port : "NULL");
}

bool isDerivedLabel(char *middleSegment) {
	struct nodeTL *temp = headTL;
	
    
    while (temp) {
        char myLabelCopy[50];
        strcpy(myLabelCopy, temp->tier);

        char *tokens[4];  // Max 4 segments
        int count = 0;

        // Tokenize the label
        char *saveptr;
        char *token = strtok_r(myLabelCopy, ".", &saveptr);
        while (token) {
            tokens[count++] = token;
            token = strtok_r(NULL, ".", &saveptr);
        }

        // Extract myMiddleSegment (everything except first and last)
        static char myMiddleBuffer[50];  
        myMiddleBuffer[0] = '\0';  

        // Handle the case where the label has 2 segments (e.g., "1.2")
        if (count == 2) {  // "tier.middle"
            strcpy(myMiddleBuffer, tokens[1]);  // Use only the second token (middle)
        } 
        else if (count == 3) {  // "tier.middle.port"
            snprintf(myMiddleBuffer, sizeof(myMiddleBuffer), "%s.%s", tokens[1], tokens[2]);
        } 
        else if (count == 4) {  // "tier.middle1.middle2.port"
            snprintf(myMiddleBuffer, sizeof(myMiddleBuffer), "%s.%s", tokens[1], tokens[2]);
        }

        char *myMiddleSegment = myMiddleBuffer;

        // Debugging output
        printf("Comparing -> Received Middle Segment: %s, My Middle Segment: %s\n",
               middleSegment ? middleSegment : "NULL",
               myMiddleSegment ? myMiddleSegment : "NULL");

        if (middleSegment && myMiddleSegment && strcmp(middleSegment, myMiddleSegment) == 0) {
            return true;
        }
        temp = temp->next;
    }
    return false;
}

bool isDerivedFrom(char *receivedLabel) {
    struct nodeTL *temp = headTL;
    
    while (temp) {

		if (strlen(temp->tier) == strlen(receivedLabel)) {
            temp = temp->next;
            continue;
        }

        char myLabelCopy[50];
        strcpy(myLabelCopy, temp->tier);

        char *tokens[4];  // Max 4 segments
        int count = 0;

        // Tokenize my label
        char *saveptr;
        char *token = strtok_r(myLabelCopy, ".", &saveptr);
        while (token) {
            tokens[count++] = token;
            token = strtok_r(NULL, ".", &saveptr);
        }

        // Ensure my label has at least 4 segments
        if (count < 4) {
            temp = temp->next;
            continue;
        }

        // Extract middle two segments of my label
        static char myMiddleBuffer[50];  
        snprintf(myMiddleBuffer, sizeof(myMiddleBuffer), "%s.%s", tokens[1], tokens[2]);
        char *myMiddleSegment = myMiddleBuffer;

        // Tokenize received label
        char receivedCopy[50];
        strcpy(receivedCopy, receivedLabel);

        char *receivedTokens[4];  
        int receivedCount = 0;

        char *saveptr2;
        char *receivedToken = strtok_r(receivedCopy, ".", &saveptr2);
        while (receivedToken) {
            receivedTokens[receivedCount++] = receivedToken;
            receivedToken = strtok_r(NULL, ".", &saveptr2);
        }

        // Ensure received label has at least 2 segments
        if (receivedCount < 2) {
            temp = temp->next;
            continue;
        }

        // Extract last two segments of received label
        static char receivedLastBuffer[50];  
        snprintf(receivedLastBuffer, sizeof(receivedLastBuffer), "%s.%s", receivedTokens[receivedCount - 2], receivedTokens[receivedCount - 1]);
        char *receivedLastSegment = receivedLastBuffer;

        // Debugging output
        printf("\nComparing -> Received Last Two: %s (%s), My Middle Two: %s (%s)\n",
               receivedLastSegment ? receivedLastSegment : "NULL", receivedLastSegment ? receivedLabel : "NULL",
               myMiddleSegment ? myMiddleSegment : "NULL", myMiddleSegment ? myLabelCopy : "NULL");

        if (receivedLastSegment && myMiddleSegment && strcmp(receivedLastSegment, myMiddleSegment) == 0) {
            return true;
        }

        temp = temp->next;
    }

    return false;
}

bool checkIfChild(char* label, char* neighbor)
{
    int l1, l2, l3, l4;
    int n1, n2, n3;

    // Parse label (expecting 4 segments)
    if (sscanf(label, "%d.%d.%d.%d", &l1, &l2, &l3, &l4) != 4)
        return false;

    // Parse neighbor (expecting 3 segments)
    if (sscanf(neighbor, "%d.%d.%d", &n1, &n2, &n3) != 3)
        return false;

    // Check if middle two segments of label
    // match last two segments of neighbor

    if (l2 == n2 && l3 == n3)
        return true;

    return false;
}

    *totalDeletedCount = 0;

    // Tejas addition - Fix 9: Compute prefix to identify labels assigned by THIS C-node
    // Format: "2.<our_addr_suffix>." e.g., "2.1." for C1 (label 1.1), "2.2." for C2 (label 1.2)
    // When a received delete message contains a label matching our prefix, it means
    // our connection to that D-node is broken. ALL other labels on the same port must
    // also be removed from our neighbor table and their grandchildren from routing table.
    // Problem solved: When D1 loses connection to C1, C1 receives delete for 2.1.1 from peers.
    // C1 must also remove 2.2.1 (D1's label from C2) from its neighbor table because
    // C1 can no longer reach D1 at all. Peer C-nodes (C2) receiving the same message
    // do NOT trigger port cleanup because 2.1.1 does not match C2's prefix "2.2.".
    // Does NOT impact IF_TIMER_DOWN: by the time forwarded messages arrive back at C1,
    // the outer delete() loop has already cleaned all entries on the failed port.
    char* myTierAddress = getTierInfo();
    char myPrefix[10];
    snprintf(myPrefix, sizeof(myPrefix), "2.%s.", myTierAddress + 2);
    char portsToClean[5][20];
    int portsToCleanCount = 0;

    // **Step 1: Process all labels in Neighbor Table**
    // Tejas addition - Fix 8: Use exact match only (removed checkIfChild)
    // Previously this also had: || checkIfChild(label, currentNeighbor->tier)
    // which caused tier 3 labels (e.g., 3.3.3.4) to incorrectly remove tier 2 labels
    // (e.g., 2.3.3) from neighbor table during A-node failures.
    // Problem solved: A-node failure message has only tier 3 labels. With checkIfChild,
    // C-nodes would infer tier 2 parent labels and remove them, incorrectly losing
    // D-node neighbor entries for still-alive D-nodes. With exact match only, tier 3
    // labels don't match any tier 2 neighbor entries, so neighbor table stays untouched.
    // D-node failure message includes tier 2 labels explicitly (Fix 6), so exact match works.
    struct nodeHL *currentNeighbor, *prevNeighbor;
    for (int i = 0; i < labelCount; i++) {
        char* label = labels[i];
        currentNeighbor = headHL;
        prevNeighbor = NULL;

        while (currentNeighbor != NULL) {

            if (strcmp(currentNeighbor->tier, label) == 0)
            {
                printf("\nRemoving neighbor: %s\n", currentNeighbor->tier);
                isRemoved = true;

                // Tejas addition - Fix 9: If this is OUR label for the D-node, save the port
                // Our connection to this D-node is broken, all labels on this port are stale
                if (strncmp(label, myPrefix, strlen(myPrefix)) == 0) {
                    bool portAlreadySaved = false;
                    for (int p = 0; p < portsToCleanCount; p++) {
                        if (strcmp(portsToClean[p], currentNeighbor->port) == 0) {
                            portAlreadySaved = true;
                            break;
                        }
                    }
                    if (!portAlreadySaved && portsToCleanCount < 5) {
                        printf("Own label %s deleted, will clean all entries on port %s\n",
                               label, currentNeighbor->port);
                        strcpy(portsToClean[portsToCleanCount], currentNeighbor->port);
                        portsToCleanCount++;
                    }
                }

                // Store deleted label
                deletedLabels[*totalDeletedCount] = strdup(currentNeighbor->tier);
                if (deletedLabels[*totalDeletedCount] != NULL)
                    (*totalDeletedCount)++;

                add_neighbor_blacklist(currentNeighbor->tier);

                // Remove from list
                if (prevNeighbor == NULL)
                    headHL = currentNeighbor->next;
                else
                    prevNeighbor->next = currentNeighbor->next;

                struct nodeHL* temp = currentNeighbor;
                currentNeighbor = currentNeighbor->next;
                free(temp);

                continue;  // continue checking rest of neighbors
            }

            prevNeighbor = currentNeighbor;
            currentNeighbor = currentNeighbor->next;
        }
    }

    // Tejas addition - Fix 9: Port-based cleanup for own labels
    // Remove ALL remaining entries on ports where our own labels were deleted.
    // These are D-node's other labels from other C-nodes (e.g., 2.2.1 from C2
    // when 2.1.1 from C1 was deleted). NOT added to deletedLabels (not forwarded
    // to peers) because the D-node is still alive through other C-nodes.
    for (int p = 0; p < portsToCleanCount; p++) {
        currentNeighbor = headHL;
        prevNeighbor = NULL;
        while (currentNeighbor != NULL) {
            if (strcmp(currentNeighbor->port, portsToClean[p]) == 0) {
                printf("Port cleanup: removing %s on port %s (local only)\n",
                       currentNeighbor->tier, portsToClean[p]);
                isRemoved = true;
                add_neighbor_blacklist(currentNeighbor->tier);

                if (prevNeighbor == NULL)
                    headHL = currentNeighbor->next;
                else
                    prevNeighbor->next = currentNeighbor->next;

                struct nodeHL* temp = currentNeighbor;
                currentNeighbor = currentNeighbor->next;
                free(temp);
                continue;
            }
            prevNeighbor = currentNeighbor;
            currentNeighbor = currentNeighbor->next;
        }
    }

    printNeighbourTable(isRemoved);

    // **Step 2: Process all labels in LL Table**
    struct addr_tuple* currentLL, *prevLL;
    for (int i = 0; i < labelCount; i++) {
        char* label = labels[i];
        currentLL = tablehead;
        prevLL = NULL;

        while (currentLL != NULL) {
            if (strcmp(currentLL->tier_addr, label) == 0 || (strlen(label) != 7 && checkIfGrandParent(currentLL->tier_addr, label))) {
                printf("\nFound grandchild in Ip to Label Table, deleting: %s\n", currentLL->tier_addr);

                // Resize array if needed
                if (*totalDeletedCount >= allocatedSize) {
                    allocatedSize *= 2;
                    char** temp = realloc(deletedLabels, allocatedSize * sizeof(char*));
                    if (temp == NULL) {
                        perror("Failed to reallocate memory for deleted labels");
                        break;
                    }
                    deletedLabels = temp;
                }

                // Store label
                deletedLabels[*totalDeletedCount] = strdup(currentLL->tier_addr);
                if (deletedLabels[*totalDeletedCount] == NULL) {
                    perror("Failed to duplicate label");
                } else {
                    (*totalDeletedCount)++;
                }

                // Remove from LL Table
                if (prevLL == NULL) {
                    tablehead = currentLL->next;
                } else {
                    prevLL->next = currentLL->next;
                }
                free(currentLL);
            }
            prevLL = currentLL;
            currentLL = currentLL->next;
        }
    }

    // Grandchildren of port-cleaned labels are NOT removed from routing table —
    // those routes are still reachable through other C-nodes.

    print_entries_LL();

    if (*totalDeletedCount == 0) {
        free(deletedLabels);
        return NULL;
    }

    return deletedLabels;
}


char** deleteLabelsForFailedInterface(char* failedInterface) {
	printf("\ninside deleteLabelsForFailedInterface\n");
    if (failedInterface == NULL || headHL == NULL || tablehead == NULL || headInterfaceLabel == NULL) {
        return NULL;
    }

    // **Step 1: Find Child Labels for the Failed Interface**
    struct childInterfaceToLabelList* currentInterfaceLabel = headInterfaceLabel;
    while (currentInterfaceLabel != NULL) {
        if (strcmp(currentInterfaceLabel->interface, failedInterface) == 0) {
            break;
        }
        currentInterfaceLabel = currentInterfaceLabel->next;
    }

    if (currentInterfaceLabel == NULL || currentInterfaceLabel->labelList == NULL) {
        printf("\nNo labels found for failed interface: %s\n", failedInterface);
        return NULL;
    }

    // **Step 2: Count Child Labels for Memory Allocation**
    int labelCount = 0;
    struct labels* labelNode = currentInterfaceLabel->labelList;
    while (labelNode != NULL) {
        labelCount++;
        labelNode = labelNode->next;
    }

    // Allocate memory for deleted labels
    char** deletedLabels = malloc(labelCount * 2 * sizeof(char*)); // Double size for children + grandchildren
    if (deletedLabels == NULL) {
        perror("Failed to allocate memory for deleted labels");
        return NULL;
    }

    int deletedIndex = 0;
    labelNode = currentInterfaceLabel->labelList;

    // **Step 3: Delete Child Labels from Neighbor Table**
    while (labelNode != NULL) {
        char* childLabel = labelNode->label;
        struct nodeHL *currentNeighbor = headHL, *prevNeighbor = NULL;

        while (currentNeighbor != NULL) {
            if (strcmp(currentNeighbor->tier, childLabel) == 0) {
                printf("\nRemoving child label from Neighbor Table: %s\n", childLabel);

                deletedLabels[deletedIndex] = strdup(childLabel);
                if (deletedLabels[deletedIndex] == NULL) {
                    perror("Failed to duplicate label");
                } else {
                    deletedIndex++;
                }

                // Remove from Neighbor Table
                if (prevNeighbor == NULL) {
                    headHL = currentNeighbor->next;
                } else {
                    prevNeighbor->next = currentNeighbor->next;
                }
                free(currentNeighbor);
                break;
            }
            prevNeighbor = currentNeighbor;
            currentNeighbor = currentNeighbor->next;
        }
        labelNode = labelNode->next;
    }

    printNeighbourTable((deletedIndex>0) ? 1:0);

    // **Step 4: Find and Delete Grandchildren from LL Table**
    labelNode = currentInterfaceLabel->labelList;
    while (labelNode != NULL) {
        char* childLabel = labelNode->label;
        struct addr_tuple* currentLL = tablehead;
        struct addr_tuple* prevLL = NULL;

        while (currentLL != NULL) {
            if (strlen(childLabel) != 7 && checkIfGrandParent(currentLL->tier_addr, childLabel)) {
                printf("\nFound grandchild in LL Table, deleting: %s\n", currentLL->tier_addr);

                deletedLabels[deletedIndex] = strdup(currentLL->tier_addr);
                if (deletedLabels[deletedIndex] == NULL) {
                    perror("Failed to duplicate label");
                } else {
                    deletedIndex++;
                }

                // Remove from LL Table
                if (prevLL == NULL) {
                    tablehead = currentLL->next;
                } else {
                    prevLL->next = currentLL->next;
                }
                free(currentLL);
            }
            prevLL = currentLL;
            currentLL = currentLL->next;
        }
        labelNode = labelNode->next;
    }

    print_entries_LL();

    // If no labels were deleted, free allocated memory and return NULL
	
    if (deletedIndex == 0) {
        	
        for (int i = 0; i < labelCount; i++) {
        	if (deletedLabels[i] != NULL) {
            	free(deletedLabels[i]);
            	deletedLabels[i] = NULL; // Set to NULL after freeing to avoid double free
        }
    

        return NULL;
    }
	}
    // Null terminate the list for easy processing
    deletedLabels[deletedIndex] = NULL;
	
    return deletedLabels;
}

void msg_typ_ip_add(char buffer[2048], unsigned char *ethhead, char recvOnEtherPort[5], struct sockaddr_ll src_addr, int *n) {
    printf("\nReceived MESSAGE_TYPE_PUBLISH_IP_ADD on : %s\n", recvOnEtherPort);

    if (myTierValue == 3) {
        printf("\nMESSAGE_TYPE_PUBLISH_IP_ADD received at destination\n");

        int lengthIndex = 16;
        int lengthOfTierAddrTemp = ethhead[lengthIndex];
        unsigned char tierAddrTemp[lengthOfTierAddrTemp + 1];
        memset(tierAddrTemp, '\0', sizeof(tierAddrTemp));
        lengthIndex++;
        memcpy(tierAddrTemp, &ethhead[lengthIndex], lengthOfTierAddrTemp);
        lengthIndex += lengthOfTierAddrTemp;
        printf("\nGot in response : %s", tierAddrTemp);

        uint8_t ipLen = ethhead[lengthIndex];
        struct in_addr ipAddr;
        lengthIndex++;
        memcpy(&ipAddr, &ethhead[lengthIndex], sizeof(struct in_addr));
        lengthIndex += sizeof(struct in_addr);
        uint8_t cidr = ethhead[lengthIndex];
        lengthIndex++;

        // Check if the label is already processed
        if (is_label_processed((char *)tierAddrTemp)) {
            printf("\nLabel %s already processed, removing from the list.\n", tierAddrTemp);
            remove_processed_label((char *)tierAddrTemp);
            return;
        }

        if (strcmp(ip_add, inet_ntoa(ipAddr)) == 0 || searchIPinMyMap(inet_ntoa(ipAddr)) == 0) {
            return;
        }

        printf("\nipLen :%u IPAddr: %s cidr: %u\n", ipLen, inet_ntoa(ipAddr), cidr);

        struct addr_tuple *a = (struct addr_tuple *)malloc(sizeof(struct addr_tuple));
        memset(a, '\0', sizeof(struct addr_tuple));
        strcpy(a->tier_addr, (char *)tierAddrTemp);

        if (find_entry_LL(&ipAddr, (char *)tierAddrTemp) == NULL) {
            a->if_index = src_addr.sll_ifindex;
            a->isNew = true;
            memcpy(&a->ip_addr, &ipAddr, sizeof(struct in_addr));
            a->cidr = cidr;
            churn_flag = true;
            add_entry_LL(a);
            print_entries_LL();
        }
    } else {
        printf("\nMESSAGE_TYPE_PUBLISH_IP_ADD sent to my children to update\n");

        unsigned char *headWithPayload;
        int packetSize = *n - 14;
        headWithPayload = (unsigned char *)malloc(packetSize);
        memset(headWithPayload, '\0', packetSize);
        memcpy(headWithPayload, &buffer[14], packetSize);

        setInterfaces();
        for (int loopCounter2 = 1; loopCounter2 < interfaceListSize; loopCounter2++) {
            int port_number = matchPort(interfaceList[loopCounter2]);
            if (port_number == 2) {
                printf("\nSending MESSAGE_TYPE_PUBLISH_IP_ADD message to - %s\n", interfaceList[loopCounter2]);
                endNetworkSend(interfaceList[loopCounter2], headWithPayload, packetSize);
            }
        }

        free(headWithPayload);
        freeInterfaces();
        interfaceListSize = 0;
    }
}
